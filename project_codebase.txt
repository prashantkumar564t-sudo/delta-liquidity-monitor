"""
generate_report.py
Generates summary report specifically for the LATEST collection session.
"""

import csv
import json
from datetime import datetime
from pathlib import Path


def load_csv(path):
    if not path.exists():
        return []
    with path.open("r", encoding="utf-8") as f:
        return list(csv.DictReader(f))


def parse_iso(ts_str):
    try:
        return datetime.fromisoformat(ts_str.replace("Z", "+00:00"))
    except Exception:
        return None


def get_latest_session_data(all_data, gap_threshold_seconds=10):
    """Detects where the latest run started by checking time gaps between snapshots."""
    if not all_data:
        return []

    # Iterate backwards to find when the last run started
    session_start_idx = 0
    for i in range(len(all_data) - 1, 0, -1):
        t_curr = parse_iso(all_data[i]["timestamp_utc"])
        t_prev = parse_iso(all_data[i - 1]["timestamp_utc"])
        
        if t_curr and t_prev:
            diff = (t_curr - t_prev).total_seconds()
            if diff > gap_threshold_seconds:
                # Big gap found! The latest session starts at index i
                session_start_idx = i
                break

    return all_data[session_start_idx:]


def generate_summary():
    root = Path(__file__).resolve().parent
    metrics_file = root / "data" / "analysis" / "metrics.csv"
    resource_file = root / "logs" / "system_resources.csv"

    all_metrics = load_csv(metrics_file)
    resource_data = load_csv(resource_file)

    if not all_metrics:
        print("[!] No metrics data found. Run launcher.py first.")
        return

    # सिर्फ सबसे ताज़ा (Current) रन का डेटा लें
    latest_metrics = get_latest_session_data(all_metrics)

    t_start = parse_iso(latest_metrics[0]["timestamp_utc"])
    t_end = parse_iso(latest_metrics[-1]["timestamp_utc"])
    
    if t_start and t_end:
        duration_sec = int((t_end - t_start).total_seconds())
        duration_str = f"{duration_sec // 60}m {duration_sec % 60}s"
    else:
        duration_str = "5m 00s"

    # Price & Spread (सिर्फ ताज़ा रन के)
    mid_prices = [float(row["mid_price"]) for row in latest_metrics]
    spreads = [float(row["spread"]) for row in latest_metrics]
    spread_pcts = [float(row["spread_percent"]) for row in latest_metrics]

    # Liquidity
    bid_volumes_usd = [float(row["bid_notional_usd"]) for row in latest_metrics]
    ask_volumes_usd = [float(row["ask_notional_usd"]) for row in latest_metrics]
    imbalances = [float(row["imbalance"]) for row in latest_metrics]

    # Slippage
    buy_slippages = []
    sell_slippages = []
    for row in latest_metrics:
        try:
            buy_info = json.loads(row.get("buy_slippage_json", "{}"))
            sell_info = json.loads(row.get("sell_slippage_json", "{}"))
            buy_slippages.append(float(buy_info.get("slippage_percent", 0.0)))
            sell_slippages.append(float(sell_info.get("slippage_percent", 0.0)))
        except Exception:
            pass

    # RAM
    ram_usages = [float(row["memory_mb"]) for row in resource_data if float(row["memory_mb"]) > 0]
    peak_ram = max([float(row["peak_memory_mb"]) for row in resource_data]) if resource_data else 0.0

    print("=" * 65)
    print("      DELTA EXCHANGE BTC-USD REPORT (LATEST 5-MIN RUN)      ")
    print("=" * 65)
    print(f"Snapshots in This Run    : {len(latest_metrics)} (Total History: {len(all_metrics)})")
    print(f"Exact Run Duration       : {duration_str}")
    print(f"Session Start Time (UTC) : {latest_metrics[0]['timestamp_utc']}")
    print(f"Session End Time (UTC)   : {latest_metrics[-1]['timestamp_utc']}")
    print("-" * 65)
    print("1. PRICE & SPREAD DYNAMICS (THIS RUN):")
    print(f"   • Starting Mid Price  : ${mid_prices[0]:,.2f}")
    print(f"   • Ending Mid Price    : ${mid_prices[-1]:,.2f}")
    print(f"   • Price Movement      : ${mid_prices[-1] - mid_prices[0]:+,.2f}")
    print(f"   • Min / Max Mid Price : ${min(mid_prices):,.2f} / ${max(mid_prices):,.2f}")
    print(f"   • Average Spread      : ${sum(spreads)/len(spreads):.4f} ({sum(spread_pcts)/len(spread_pcts):.6f}%)")
    print("-" * 65)
    print("2. ORDER BOOK DEPTH & PRESSURE:")
    print(f"   • Avg Bid Depth (USD) : ${sum(bid_volumes_usd)/len(bid_volumes_usd):,.2f}")
    print(f"   • Avg Ask Depth (USD) : ${sum(ask_volumes_usd)/len(ask_volumes_usd):,.2f}")
    avg_imbalance = sum(imbalances)/len(imbalances)
    side_dominance = "Bids Dominant (Buy Pressure)" if avg_imbalance > 0 else "Asks Dominant (Sell Pressure)"
    print(f"   • Avg Order Imbalance : {avg_imbalance:+.4f} ({side_dominance})")
    print("-" * 65)
    print("3. EXECUTION SLIPPAGE ($1,000 Order):")
    if buy_slippages:
        print(f"   • Avg Buy Slippage    : {sum(buy_slippages)/len(buy_slippages):.6f}%")
    if sell_slippages:
        print(f"   • Avg Sell Slippage   : {sum(sell_slippages)/len(sell_slippages):.6f}%")
    print("-" * 65)
    print("4. SYSTEM RESOURCE PROFILER:")
    if ram_usages:
        print(f"   • Average RAM Usage   : {sum(ram_usages)/len(ram_usages):.2f} MB")
    print(f"   • Peak RAM Usage      : {peak_ram:.2f} MB")
    print("=" * 65)


if __name__ == "__main__":
    generate_summary()

"""
launcher.py
Launches the collector process and the RAM/Resource monitor concurrently.
"""

import subprocess
import sys
import time
from pathlib import Path

def main():
    python_exe = sys.executable
    root_dir = Path(__file__).resolve().parent

    main_script = root_dir / "src" / "main.py"
    monitor_script = root_dir / "monitor" / "ram_monitor.py"

    print("=" * 60)
    print("Launching Delta Liquidity Monitor & Profiler")
    print("=" * 60)

    # 1. Start Main Collector
    collector_proc = subprocess.Popen([python_exe, str(main_script)])
    print(f"[Launcher] Collector started (PID: {collector_proc.pid})")

    # 2. Start Resource Monitor with collector's PID
    monitor_proc = subprocess.Popen([python_exe, str(monitor_script), str(collector_proc.pid)])
    print(f"[Launcher] Monitor started (PID: {monitor_proc.pid})")

    try:
        # Wait for collector to complete
        collector_proc.wait()
    except KeyboardInterrupt:
        print("\n[Launcher] Shutting down...")
        collector_proc.terminate()
    finally:
        # Stop monitor process
        if monitor_proc.poll() is None:
            time.sleep(1)
            monitor_proc.terminate()

    print("[Launcher] All tasks finished successfully.")

if __name__ == "__main__":
    main()

EXCLUDE_EXTS = {'.pyc', '.png', '.jpg', '.jpeg', '.zip', '.tar', '.gz', '.db', '.sqlite3'}

def bundle_project(root_dir="."):
    with open(OUTPUT_FILE, "w", encoding="utf-8") as outfile:
        # 1. पहले प्रोजेक्ट का फाइल स्ट्रक्चर लिखें
        outfile.write("=== PROJECT STRUCTURE ===\n")
        for root, dirs, files in os.walk(root_dir):
            dirs[:] = [d for d in dirs if d not in EXCLUDE_DIRS]
            level = root.replace(root_dir, '').count(os.sep)
            indent = ' ' * 4 * level
            outfile.write(f"{indent}{os.path.basename(root)}/\n")
            sub_indent = ' ' * 4 * (level + 1)
            for f in files:
                if not any(f.endswith(ext) for ext in EXCLUDE_EXTS):
                    outfile.write(f"{sub_indent}{f}\n")

            for file in files:
                if any(file.endswith(ext) for ext in EXCLUDE_EXTS) or file == OUTPUT_FILE:
                    continue
                file_path = os.path.join(root, file)
                rel_path = os.path.relpath(file_path, root_dir)

                    outfile.write(f"Error reading file: {e}")

    print(f"प्रोजेक्ट सफलतापूर्वक बंडल हुआ: {OUTPUT_FILE}")

if __name__ == "__main__":

# Delta Exchange BTC Liquidity Monitor & Profiler

A lightweight, zero-dependency market liquidity and Level-2 order book telemetry engine for Delta Exchange BTC-USD perpetual contracts.

Engineered with Python Standard Library only, it captures high-frequency order book snapshots, models market impact & execution slippage across capital tiers, and continuously profiles memory utilization.

---

## Key Features

- **Zero External Dependencies**: Operates strictly on Python's built-in libraries (`urllib`, `ctypes`, `csv`, `json`, `time`, `pathlib`).
- **High-Frequency Snapshotting**: Sub-second REST polling with deterministic time-drift correction.
- **Microstructure Telemetry**:
  - Tight tick spread analysis and bid/ask contract depth.
  - Real-time order book imbalance index ($[-1.0, +1.0]$) to gauge directional order flow pressure.
- **Strict Budget Execution Modeling**: Multi-level ladder walking simulator calculating effective fill prices, nominal slippage, and taker fees.
- **Isolated Resource Profiling**: Concurrently tracks Python process RAM and working set memory with zero external profiling overhead (<27 MB RAM footprint).

"""
monitor/ram_monitor.py
Tracks RAM of target process using Standard Library only (Windows-safe).
"""

import sys
import time
import os
import csv
from datetime import datetime, timezone
from pathlib import Path


def is_process_running(pid):
    """Check if process is alive without sending signals (Windows-safe)."""
    if os.name == 'nt':
        import ctypes
        from ctypes import wintypes
        PROCESS_QUERY_LIMITED_INFORMATION = 0x1000
        handle = ctypes.windll.kernel32.OpenProcess(PROCESS_QUERY_LIMITED_INFORMATION, False, pid)
        if not handle:
            return False
        exit_code = wintypes.DWORD()
        ctypes.windll.kernel32.GetExitCodeProcess(handle, ctypes.byref(exit_code))
        ctypes.windll.kernel32.CloseHandle(handle)
        return exit_code.value == 259  # 259 = STILL_ACTIVE
    else:
        try:
            os.kill(pid, 0)
            return True
        except (OSError, ProcessLookupError):
            return False


def get_process_memory_mb(pid):
    try:
        if os.name == 'nt':
            import ctypes
            from ctypes import wintypes

            class PROCESS_MEMORY_COUNTERS(ctypes.Structure):
                _fields_ = [
                    ('cb', wintypes.DWORD),
                    ('PageFaultCount', wintypes.DWORD),
                    ('PeakWorkingSetSize', ctypes.c_size_t),
                    ('WorkingSetSize', ctypes.c_size_t),
                    ('QuotaPeakPagedPoolUsage', ctypes.c_size_t),
                    ('QuotaPagedPoolUsage', ctypes.c_size_t),
                    ('QuotaPeakNonPagedPoolUsage', ctypes.c_size_t),
                    ('QuotaNonPagedPoolUsage', ctypes.c_size_t),
                    ('PagefileUsage', ctypes.c_size_t),
                    ('PeakPagefileUsage', ctypes.c_size_t),
                ]

            PROCESS_QUERY_INFORMATION = 0x0400
            PROCESS_VM_READ = 0x0010
            handle = ctypes.windll.kernel32.OpenProcess(
                PROCESS_QUERY_INFORMATION | PROCESS_VM_READ, False, pid
            )
            if not handle:
                return 0.0

            counters = PROCESS_MEMORY_COUNTERS()
            counters.cb = ctypes.sizeof(PROCESS_MEMORY_COUNTERS)
            ctypes.windll.psapi.GetProcessMemoryInfo(
                handle, ctypes.byref(counters), counters.cb
            )
            ctypes.windll.kernel32.CloseHandle(handle)
            return counters.WorkingSetSize / (1024 * 1024)
        else:
            with open(f"/proc/{pid}/status", "r") as f:
                for line in f:
                    if line.startswith("VmRSS:"):
                        return float(line.split()[1]) / 1024.0
    except Exception:
        return 0.0
    return 0.0


def monitor(target_pid, interval=1.0):
    project_root = Path(__file__).resolve().parent.parent
    log_file = project_root / "logs" / "system_resources.csv"
    log_file.parent.mkdir(parents=True, exist_ok=True)

    file_exists = log_file.exists() and log_file.stat().st_size > 0

    try:
        with log_file.open("a", newline="", encoding="utf-8") as f:
            writer = csv.writer(f)
            if not file_exists:
                writer.writerow(["timestamp_utc", "target_pid", "memory_mb", "peak_memory_mb"])

            peak_memory = 0.0
            while True:
                # सुरक्षित विंडोज़ प्रोसेस चेक
                if not is_process_running(target_pid):
                    break

                mem = get_process_memory_mb(target_pid)
                if mem > peak_memory:
                    peak_memory = mem

                writer.writerow([
                    datetime.now(timezone.utc).isoformat(),
                    target_pid,
                    f"{mem:.2f}",
                    f"{peak_memory:.2f}"
                ])
                f.flush()
                time.sleep(interval)
    except KeyboardInterrupt:
        pass


if __name__ == "__main__":
    if len(sys.argv) > 1:
        pid = int(sys.argv[1])
        monitor(pid)

import json
import urllib.request
import urllib.error


class DeltaAPI:
    """Lightweight Delta Exchange REST API client."""

    BASE_URL = "https://api.india.delta.exchange"

    def __init__(self, timeout=10):
        self.timeout = timeout

    def _get(self, endpoint):
        """Send a public GET request and return parsed JSON."""

        url = f"{self.BASE_URL}{endpoint}"

        request = urllib.request.Request(
            url,
            headers={
                "User-Agent": "Delta-Liquidity-Monitor/1.0",
                "Accept": "application/json",
            },
            method="GET",
        )

        try:
            with urllib.request.urlopen(
                request,
                timeout=self.timeout
            ) as response:

                status_code = response.status
                raw_data = response.read()

                if status_code != 200:
                    raise RuntimeError(
                        f"Unexpected HTTP status: {status_code}"
                    )

                data = json.loads(raw_data.decode("utf-8"))

                if not data.get("success", False):
                    raise RuntimeError(
                        f"Delta API error: {data}"
                    )

                return data.get("result")

        except urllib.error.HTTPError as error:
            raise RuntimeError(
                f"HTTP Error {error.code}: {error.reason}"
            ) from error

        except urllib.error.URLError as error:
            raise RuntimeError(
                f"Connection Error: {error.reason}"
            ) from error

        except TimeoutError as error:
            raise RuntimeError(
                "API request timed out"
            ) from error

        except json.JSONDecodeError as error:
            raise RuntimeError(
                "Invalid JSON response from Delta Exchange"
            ) from error

    def get_product(self, symbol="BTCUSD"):
        """Fetch product specifications."""

        return self._get(
            f"/v2/products/{symbol}"
        )

    def get_orderbook(self, symbol="BTCUSD", depth=5):
        """Fetch Level 2 order book snapshot."""

        return self._get(
            f"/v2/l2orderbook/{symbol}?depth={depth}"
        )

"""
src/main.py
Main market liquidity collector and analyzer.
"""

import time
import sys
from datetime import datetime, timezone
from pathlib import Path

sys.path.insert(0, str(Path(__file__).resolve().parent.parent))

from src import config
from src.delta_api import DeltaAPI
from src.orderbook import OrderBookAnalyzer
from src.slippage import SlippageCalculator
from src.storage import CSVStorage


def run_collector(duration_seconds=config.COLLECTION_DURATION, interval=config.SNAPSHOT_INTERVAL):
    # वेरिएबल्स को शुरू में ही इनिशियलाइज़ करें
    snapshot_count = 0
    errors_count = 0

    print("=" * 60)
    print(f"Starting Delta Liquidity Collector: {config.SYMBOL}")
    print(f"Duration: {duration_seconds}s | Interval: {interval}s")
    print("=" * 60)

    try:
        api = DeltaAPI(timeout=config.API_TIMEOUT)
        book_analyzer = OrderBookAnalyzer(contract_value=config.CONTRACT_VALUE)
        slippage_calc = SlippageCalculator(
            contract_value=config.CONTRACT_VALUE,
            taker_fee=config.TAKER_FEE
        )
        storage = CSVStorage()

        start_time = time.monotonic()

        while True:
            elapsed = time.monotonic() - start_time
            if elapsed >= duration_seconds:
                print("\n[+] Configured collection duration completed.")
                break

            loop_start = time.monotonic()
            timestamp_utc = datetime.now(timezone.utc).isoformat()

            try:
                orderbook = api.get_orderbook(symbol=config.SYMBOL, depth=config.DEPTH)
                if not orderbook:
                    raise RuntimeError("Empty response received from orderbook API")

                storage.write_raw_orderbook(
                    orderbook=orderbook,
                    symbol=config.SYMBOL,
                    timestamp=timestamp_utc
                )

                metrics = book_analyzer.analyze(orderbook)

                buy_slip = slippage_calc.simulate(
                    orderbook, order_usd=config.PRIMARY_SIMULATION_USD, side="buy"
                )
                sell_slip = slippage_calc.simulate(
                    orderbook, order_usd=config.PRIMARY_SIMULATION_USD, side="sell"
                )

                storage.write_metrics(
                    metrics=metrics,
                    slippage_results={"buy": buy_slip, "sell": sell_slip},
                    timestamp=timestamp_utc
                )

                snapshot_count += 1
                sys.stdout.write(
                    f"\r[{snapshot_count:04d}] Mid: ${metrics['mid_price']:.2f} | "
                    f"Spread: ${metrics['spread']:.2f} ({metrics['spread_percent']:.4f}%) | "
                    f"Buy Slip: {buy_slip['slippage_percent']:.4f}% | "
                    f"Elapsed: {int(elapsed)}s/{duration_seconds}s"
                )
                sys.stdout.flush()

            except Exception as err:
                errors_count += 1
                print(f"\n[!] Snapshot warning: {err}")

            iteration_time = time.monotonic() - loop_start
            sleep_time = max(0.0, interval - iteration_time)
            time.sleep(sleep_time)

    except KeyboardInterrupt:
        print("\n[!] Collection stopped by user.")

    print("\n" + "=" * 60)
    print(f"Collection Complete!")
    print(f"Total Snapshots: {snapshot_count}")
    print(f"Errors/Skipped: {errors_count}")
    print("=" * 60)


if __name__ == "__main__":
    run_collector()

class OrderBookAnalyzer:
    """Analyze Delta Exchange Level 2 order book snapshots."""

    def __init__(self, contract_value=0.001):
        self.contract_value = contract_value

    def analyze(self, orderbook):
        bids = orderbook.get("buy", [])
        asks = orderbook.get("sell", [])

        if not bids or not asks:
            raise ValueError("Order book is missing bid or ask levels")

        best_bid = float(bids[0]["price"])
        best_ask = float(asks[0]["price"])

        if best_bid <= 0 or best_ask <= 0:
            raise ValueError("Invalid order book prices")

        mid_price = (best_bid + best_ask) / 2
        spread = best_ask - best_bid
        spread_percent = (spread / mid_price) * 100

        bid_contracts = sum(
            int(level["size"]) for level in bids
        )

        ask_contracts = sum(
            int(level["size"]) for level in asks
        )

        bid_btc = bid_contracts * self.contract_value
        ask_btc = ask_contracts * self.contract_value

        bid_notional = bid_btc * best_bid
        ask_notional = ask_btc * best_ask

        total_depth = bid_contracts + ask_contracts

        imbalance = (
            (bid_contracts - ask_contracts) / total_depth
            if total_depth else 0
        )

        return {
            "best_bid": best_bid,
            "best_ask": best_ask,
            "mid_price": mid_price,
            "spread": spread,
            "spread_percent": spread_percent,
            "bid_contracts": bid_contracts,
            "ask_contracts": ask_contracts,
            "bid_btc": bid_btc,
            "ask_btc": ask_btc,
            "bid_notional_usd": bid_notional,
            "ask_notional_usd": ask_notional,
            "imbalance": imbalance,
        }

from decimal import Decimal, ROUND_FLOOR


class SlippageCalculator:
    """Simulate market execution with a strict USD budget."""

    def __init__(self, contract_value=0.001, taker_fee=0.0005):
        self.contract_value = Decimal(str(contract_value))
        self.taker_fee = Decimal(str(taker_fee))

    def simulate(self, orderbook, order_usd, side="buy"):
        budget = Decimal(str(order_usd))

        if budget <= 0:
            raise ValueError("Order size must be greater than zero")

        if side not in ("buy", "sell"):
            raise ValueError("Side must be 'buy' or 'sell'")

        levels = orderbook.get(
            "sell" if side == "buy" else "buy", []
        )

        if not levels:
            raise ValueError("Order book side is empty")

        best_price = Decimal(str(levels[0]["price"]))

        if best_price <= 0:
            raise ValueError("Invalid best price")

        remaining_usd = budget
        executed_contracts = 0
        executed_notional = Decimal("0")
        executed_levels = 0

        for level in levels:
            price = Decimal(str(level["price"]))
            available = int(level["size"])

            if price <= 0 or available < 0:
                raise ValueError("Invalid order book level")

            contract_notional = price * self.contract_value

            affordable = int(
                (remaining_usd / contract_notional).to_integral_value(
                    rounding=ROUND_FLOOR
                )
            )

            fill = min(available, affordable)

            if fill <= 0:
                continue

            fill_notional = fill * contract_notional

            executed_contracts += fill
            executed_notional += fill_notional
            remaining_usd -= fill_notional
            executed_levels += 1

            if remaining_usd == 0:
                break

        if executed_contracts == 0:
            raise ValueError("No contracts affordable or available")

        average_price = (
            executed_notional /
            (executed_contracts * self.contract_value)
        )

        if side == "buy":
            slippage = average_price - best_price
        else:
            slippage = best_price - average_price

        slippage_percent = slippage / best_price * 100
        fee = executed_notional * self.taker_fee

        return {
            "side": side,
            "requested_usd": budget,
            "executed_contracts": executed_contracts,
            "executed_btc": executed_contracts * self.contract_value,
            "executed_notional_usd": executed_notional,
            "average_execution_price": average_price,
            "best_price": best_price,
            "slippage_usd": slippage,
            "slippage_percent": slippage_percent,
            "taker_fee_usd": fee,
            "unfilled_usd": remaining_usd,
            "fully_executed": remaining_usd == 0,
            "levels_used": executed_levels,
        }

"""
src/storage.py
Store raw order book snapshots and analysis metrics in CSV.

import csv
import json
from datetime import datetime, timezone
from pathlib import Path


class CSVStorage:
    """Store raw order book snapshots and analysis metrics in CSV."""

    def __init__(self, project_root=None):
        if project_root is None:
            project_root = Path(__file__).resolve().parent.parent

        self.project_root = Path(project_root)

        self.raw_path = (
            self.project_root / "data" / "raw" / "orderbook.csv"
        )

        self.metrics_path = (
            self.project_root / "data" / "analysis" / "metrics.csv"
        )

        self.raw_path.parent.mkdir(parents=True, exist_ok=True)
        self.metrics_path.parent.mkdir(parents=True, exist_ok=True)

        self.raw_fields = [
            "timestamp_utc",
            "symbol",
            "exchange_timestamp",
            "bids_json",
            "asks_json",
        ]

        # यहां buy_slippage_json और sell_slippage_json जोड़ दिया गया है
        self.metric_fields = [
            "timestamp_utc",
            "best_bid",
            "best_ask",
            "mid_price",
            "spread",
            "spread_percent",
            "bid_contracts",
            "ask_contracts",
            "bid_btc",
            "ask_btc",
            "bid_notional_usd",
            "ask_notional_usd",
            "imbalance",
            "buy_slippage_json",
            "sell_slippage_json",
        ]

    @staticmethod
    def _timestamp():
        return datetime.now(timezone.utc).isoformat()

    @staticmethod
    def _write_row(path, fields, row):
        file_exists = path.exists() and path.stat().st_size > 0

        with path.open(
            "a",
            newline="",
            encoding="utf-8"
        ) as file:

            writer = csv.DictWriter(
                file,
                fieldnames=fields,
                extrasaction="ignore"
            )

            if not file_exists:
                writer.writeheader()

            writer.writerow(row)
            file.flush()

    def write_raw_orderbook(
        self,
        orderbook,
        symbol="BTCUSD",
        exchange_timestamp=None,
        timestamp=None
    ):
        row = {
            "timestamp_utc": timestamp or self._timestamp(),
            "symbol": symbol,
            "exchange_timestamp": exchange_timestamp,
            "bids_json": json.dumps(
                orderbook.get("buy", []),
                separators=(",", ":")
            ),
            "asks_json": json.dumps(
                orderbook.get("sell", []),
                separators=(",", ":")
            ),
        }

        self._write_row(
            self.raw_path,
            self.raw_fields,
            row
        )

    def write_metrics(
        self,
        metrics,
        slippage_results=None,
        timestamp=None
    ):
        slippage_results = slippage_results or {}

        row = {
            "timestamp_utc": timestamp or self._timestamp(),
            **metrics,
            "buy_slippage_json": json.dumps(
                slippage_results.get("buy", {}),
                default=str
            ),
            "sell_slippage_json": json.dumps(
                slippage_results.get("sell", {}),
                default=str
            ),
        }

        self._write_row(
            self.metrics_path,
            self.metric_fields,
            row
        )
